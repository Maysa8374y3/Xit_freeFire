        function confirmSimulatedPayment() {
            if (!currentOrderPayload) return;

            // Save to LocalStorage
            const purchases = JSON.parse(localStorage.getItem('bx_purchases') || '[]');
            purchases.unshift(currentOrderPayload);
            localStorage.setItem('bx_purchases', JSON.stringify(purchases));

            closeModal('pixModal');
            showToast('Pagamento confirmado via Nubank! Seu arquivo foi liberado.', 'success');

            // Automatically trigger text file blob download named xit black
            triggerSimulatedDownload(currentOrderPayload.fileName, currentOrderPayload);

            // Redirect to My Downloads
            navigateTo('my-downloads');
        }

        function submitAdminReply(ticketIndex) {
            const input = document.getElementById(`adminReplyInput_${ticketIndex}`);
            const replyText = input.value.trim();
            if (!replyText) return;

            const tickets = JSON.parse(localStorage.getItem('bx_tickets') || '[]');
            if (tickets[ticketIndex]) {
                tickets[ticketIndex].reply = replyText;
                localStorage.setItem('bx_tickets', JSON.stringify(tickets));
                showToast('Resposta do ADM registrada!', 'success');
                renderAdminDashboard();
            }
        }

        function showToast(message, type = 'info') {
            const container = document.getElementById('toastContainer');
            const toast = document.createElement('div');

            let bgColor = 'bg-purple-900 border-purple-600 text-purple-200';
            let icon = 'fa-circle-info';

            if (type === 'success') {
                bgColor = 'bg-emerald-950 border-emerald-600 text-emerald-200';
                icon = 'fa-circle-check';
            } else if (type === 'error') {
                bgColor = 'bg-red-950 border-red-600 text-red-200';
                icon = 'fa-triangle-exclamation';
            } else if (type === 'warning') {
                bgColor = 'bg-amber-950 border-amber-600 text-amber-200';
                icon = 'fa-circle-exclamation';
            }

            toast.className = `p-3.5 rounded-xl border ${bgColor} text-xs font-bold shadow-2xl flex items-center gap-2 transition transform translate-y-2 opacity-0`;
            toast.innerHTML = `<i class="fa-solid ${icon} text-base"></i> <span>${message}</span>`;

            container.appendChild(toast);

            setTimeout(() => {
                toast.classList.remove('translate-y-2', 'opacity-0');
            }, 10);

            setTimeout(() => {
                toast.classList.add('opacity-0');
                setTimeout(() => toast.remove(), 300);
            }, 3500);
        }
